## 🔧 技术实现细节

### 1. Rankings 页面筛选逻辑
```typescript
// 按模型筛选
if (filter.models) {
  query = query.in('channels.prices.models.slug', filter.models);
}

// 按标签筛选
if (filter.tags) {
  filter.tags.forEach(tag => {
    query = query.contains('features', [tag]);
  });
}

// 按核验状态筛选
if (filter.verified) {
  query = query.not('verified_at', 'is', null);
}

// 按价格/评分排序
if (filter.sortBy === 'price') {
  providers.sort((a, b) => /* 价格逻辑 */);
} else if (filter.sortBy === 'rating') {
  providers.sort((a, b) => /* 评分逻辑 */);
}
```
